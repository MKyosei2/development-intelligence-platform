cmake_minimum_required(VERSION 3.20)
project(dip_hardware_stress_agent LANGUAGES CXX)
if(NOT WIN32)
 message(STATUS "dip-hardware-stress-agent is Windows-only; no target is generated.")
 return()
endif()
add_executable(dip-hardware-stress-agent src/main.cpp)
target_compile_features(dip-hardware-stress-agent PRIVATE cxx_std_20)
target_compile_definitions(dip-hardware-stress-agent PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dip-hardware-stress-agent PRIVATE d3d11 d3dcompiler)
if(MSVC)
 target_compile_options(dip-hardware-stress-agent PRIVATE /W4 /permissive- /EHsc)
endif()
