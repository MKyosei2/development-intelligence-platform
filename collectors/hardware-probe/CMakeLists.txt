cmake_minimum_required(VERSION 3.20)
project(dip_hardware_probe LANGUAGES CXX)
if(NOT WIN32)
 message(STATUS "dip-hardware-probe GPU benchmark is Windows-only; no target is generated.")
 return()
endif()
add_executable(dip-hardware-probe src/main.cpp)
target_compile_features(dip-hardware-probe PRIVATE cxx_std_20)
target_compile_definitions(dip-hardware-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dip-hardware-probe PRIVATE d3d11 dxgi d3dcompiler)
if(MSVC)
 target_compile_options(dip-hardware-probe PRIVATE /W4 /permissive- /EHsc)
endif()
