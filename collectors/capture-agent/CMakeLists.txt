cmake_minimum_required(VERSION 3.20)
project(dip_capture_agent LANGUAGES CXX)
if(NOT WIN32)
  message(STATUS "dip-capture-agent is Windows-only; no target is generated on this platform.")
  return()
endif()
add_executable(dip-capture-agent src/main.cpp)
target_compile_features(dip-capture-agent PRIVATE cxx_std_20)
target_compile_definitions(dip-capture-agent PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dip-capture-agent PRIVATE d3d11 dxgi mfplat mfreadwrite mfuuid ole32 uuid avrt)
if(MSVC)
  target_compile_options(dip-capture-agent PRIVATE /W4 /permissive- /EHsc)
endif()
