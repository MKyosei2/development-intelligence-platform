cmake_minimum_required(VERSION 3.20)
project(dip_crash_watchdog LANGUAGES CXX)
if(NOT WIN32)
  message(STATUS "dip-crash-watchdog is Windows-only; no target is generated.")
  return()
endif()
add_executable(dip-crash-watchdog src/main.cpp)
target_compile_features(dip-crash-watchdog PRIVATE cxx_std_20)
target_compile_definitions(dip-crash-watchdog PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dip-crash-watchdog PRIVATE dbghelp)
if(MSVC)
 target_compile_options(dip-crash-watchdog PRIVATE /W4 /permissive- /EHsc)
endif()
